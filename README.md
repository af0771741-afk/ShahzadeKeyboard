              private fun mathChar(c: Char, s: Style): String {
                  if (c in 'a'..'z') return String(Character.toChars(s.lower + (c - 'a')))
                  if (c in 'A'..'Z') {
                      if (s == Style.DOUBLE) {
                          val hole = when (c) {

                              'H' -> 0x210D
                              'N' -> 0x2115
                              'P' -> 0x2119
                              'Q' -> 0x211A
                              'R' -> 0x211D
                              'Z' -> 0x2124
                              else -> 0
                          }
                          if (hole != 0) return String(Character.toChars(hole))
                      }
                      return String(Character.toChars(s.upper + (c - 'A')))
                  }
                  return c.toString()
              }

                          FrameLayout.LayoutParams.WRAP_CONTENT,
                          Gravity.TOP or Gravity.END
                      ).apply { setMargins(0, dp(2), dp(5), 0) })
                  }

                  var longFired = false
                  var cancelled = false
                  val longPress = Runnable {
                      longFired = true
                      frame.performHapticFeedback(HapticFeedbackConstants.LONG_PRESS)
                      playClick()
                      if (hint != null) commit(styled(hint))
                      afterType()
                  }
                  frame.setOnTouchListener { v, e ->
                      when (e.action) {
                          MotionEvent.ACTION_DOWN -> {
                              longFired = false
                              cancelled = false
                              v.isPressed = true
                              if (hint != null) handler.postDelayed(longPress, 380)
                          }
                          MotionEvent.ACTION_MOVE -> {
                              if (e.x < 0 || e.y < 0 || e.x > v.width || e.y > v.height) {
                                  cancelled = true
                                  v.isPressed = false
                                  handler.removeCallbacks(longPress)
                              }
                          }
                          MotionEvent.ACTION_UP -> {
                              v.isPressed = false
                              handler.removeCallbacks(longPress)
                              if (!longFired && !cancelled) {
                                  v.performHapticFeedback(HapticFeedbackConstants.KEYBOARD_TAP)
                                  playClick()
                                  commit(styled(label))
                                  afterType()
                              }
                          }
                          MotionEvent.ACTION_CANCEL -> {
                              v.isPressed = false
                              handler.removeCallbacks(longPress)
                          }
                      }
                      true
                  }
                  return frame
              }

              // حذف هوشمند: متن تزیین‌شده (کشیده، پرچم، فونت) را یک‌جا پاک می‌کند
              private fun deleteOne() {
                  val ic = currentInputConnection ?: return
                  if (!ic.getSelectedText(0).isNullOrEmpty()) {
                      sendDownUpKeyEvents(KeyEvent.KEYCODE_DEL)
                      return
                  }
                  val before = ic.getTextBeforeCursor(16, 0)?.toString() ?: ""
                  var n = 0

                  for (e in decorEmojis) {
                      val tail = tatweel + e + tatweel
                      if (before.endsWith(tail)) {
                          n = tail.length
                          if (before.length > n && isFa(before[before.length - n - 1])) n++
                          break
                      }
                  }
                  if (n == 0 && before.length >= 2 && before.endsWith(tatweel) &&
                      isFa(before[before.length - 2])
                  ) {
                      n = 2
                  }
                  if (n == 0 && before.length >= 3) {
                      val cp = before.codePointBefore(before.length)
                      if (cp in 0x1F1E6..0x1F1FF && before[before.length - 3] == '\u200C') n = 3
                  }
                  if (n == 0 && before.length >= 2 &&
                      Character.isLowSurrogate(before[before.length - 1]) &&
                      Character.isHighSurrogate(before[before.length - 2])
                  ) {
                      n = 2
                  }

                  if (n > 0) ic.deleteSurroundingText(n, 0)
                  else sendDownUpKeyEvents(KeyEvent.KEYCODE_DEL)
              }

              private fun pressEnter() {
                  val ei = currentInputEditorInfo
                  val action = (ei?.imeOptions ?: 0) and EditorInfo.IME_MASK_ACTION
                  val noEnterAction = ((ei?.imeOptions ?: 0) and EditorInfo.IME_FLAG_NO_ENTER_ACTION) != 0
                  if (!noEnterAction &&
                      action != EditorInfo.IME_ACTION_NONE &&
                      action != EditorInfo.IME_ACTION_UNSPECIFIED
                  ) {
                      currentInputConnection?.performEditorAction(action)
                  } else {
                      sendDownUpKeyEvents(KeyEvent.KEYCODE_ENTER)
                  }
              }
